def _units(text: str) -> list[tuple[str, bool]]:

def _stem(w: str) -> str:
    """Light stem: plural and -ing/-ed only, never into the root (10/16: the
    old rule turned 'operations' into 'oper', which matched 'pre-operative')."""
    w = w.lower()
    for suf in ('ies', 'ings', 'ing', 'ed', 'es', 's'):
        if w.endswith(suf) and len(w) - len(suf) >= 5:
            return w[:-len(suf)] + ('i' if suf == 'ies' else '')
    return w


def _role_months(dates: str) -> float:
    m = re.findall(r'([A-Za-z]+)\s+(\d{4})', dates or '')
    if not m:
        return 0.0
    y0, m0 = int(m[0][1]), _MONTHS.get(m[0][0].lower(), 1)
    if len(m) > 1:
        y1, m1 = int(m[1][1]), _MONTHS.get(m[1][0].lower(), 1)
    else:
        t = date.today()
        y1, m1 = t.year, t.month
    return max(0.0, (y1 - y0) * 12 + (m1 - m0) + 1)

        by_role[r['id']] = [r.get('title', '')]
    for b in bank.get('bullets', []):
        if not b.get('approved') or b.get('status') == 'retired' or b['role'] not in by_role:
            continue
        by_role[b['role']].append(b['text'] + ' ' + ' '.join(b.get('tags', [])))
        by_role[b['role']] += [v['text'] for v in b.get('variants', []) if v.get('approved')]
    loose = [i for c in bank.get('skills', []) for i in c.get('items', [])]
    loose += list(bank.get('certifications', []))
    loose += [s['text'] for s in bank.get('summaries', []) if s.get('approved')]
    return {k: ' '.join(v).lower() for k, v in by_role.items()}, ' '.join(loose).lower()


def _found(tok: str, hay: str) -> bool:
    st = _stem(tok)
    if re.search(r'\b' + re.escape(st), hay):
        return True
    return any(re.search(r'\b' + re.escape(s), hay) for s in SYNONYMS.get(tok, ()))


def requirements(jd_text: str, bank: dict) -> list[dict]:
    """Every 'N years in <specialty>' the JD asks for, with his evidence."""
    out, seen = [], set()
    role_hay, loose_hay = _haystacks(bank)
    role_months = {r['id']: _role_months(r.get('dates', '')) for r in bank.get('roles', []) if r.get('on_resume') is not False}
    for unit, soft in _units(jd_text):
        for rx in (RX_IN, RX_PRE):
            for m in rx.finditer(unit):
                phrase = _clean_phrase(m.group('phrase'))
                toks = _tokens(phrase)
                if phrase in GENERIC_PHRASES or not toks:
                    continue
                key = ' '.join(sorted(toks))
                if key in seen:
                    continue
                seen.add(key)
                n = int(m.group('n'))
                if not 1 <= n <= 25:
                    continue
                # best alternative wins: roles proving it, else skills-only, else missing words
                best = None
                for alt in _alternatives(phrase):
                    roles = [rid for rid, hay in role_hay.items() if all(_found(t, hay) for t in alt)]
                    missing = [t for t in alt if not any(_found(t, hay) for hay in list(role_hay.values()) + [loose_hay])]
                    yrs = round(sum(role_months[r] for r in roles) / 12, 1) if roles else None
                    cand = (2 if roles else 1 if not missing else 0, yrs or 0, alt, roles, missing, yrs)
                    if best is None or cand[:2] > best[:2]:
                        best = cand
                _, _, toks, roles, missing, years_have = best
                if not roles and missing:
                    years_have = 0.0
                if missing:
                    covered, why = False, 'no evidence of: ' + ', '.join(missing)
                elif years_have is None:
                    covered, why = False, 'named in skills only — no role shows the years'
                elif years_have >= n * YEARS_TOLERANCE:
                    covered, why = True, f'{years_have} years across {", ".join(roles)}'
                else:
                    covered, why = False, f'{years_have} years across {", ".join(roles)} < {n} asked'
                out.append({'years': n, 'phrase': phrase, 'required': not soft, 'tokens': toks,
                            'evidence': [f'role:{r}' for r in roles] if roles else ([] if missing else ['skills']),
                            'roles': roles, 'years_have': years_have, 'covered': covered, 'why': why,
                            'unit': unit[:160]})
    return out


def extraction_suspect(jd_text: str, reqs: list[dict]) -> str | None:
    """Why an empty tier-1 list should be read as 'extraction failed': the JD
    names years and none became a requirement, or there is hardly any JD."""
    text = html.unescape(re.sub(r'<[^>]+>', ' ', jd_text or ''))
    if len(text) < 600:
        return f'JD is {len(text)} characters — too little to extract requirements from'
    years = re.search(r'\(?\b\d{1,2}\)?\s*\+?\s*(?:-\s*\d{1,2}\s*)?(?:years?|yrs)\b', text, re.I)
    if years and not any(r.get('kind') in ('years', 'specialty') for r in reqs):
        return f'JD says "{years.group(0)}" and no years requirement was extracted'
    return None

def verdict(match: dict, score: dict | None = None) -> dict:
    block, review = [], []
    lane = match.get('lane') or {}
    reqs = match.get('requirements') or []
    if lane.get('family') == 'other':
        block.append('title is in no lane the engine applies to')
    if lane.get('seniority') == 'senior':
        block.append('senior-level title')
    t1 = [r for r in reqs if r.get('tier') == 1]
    for r in t1:
        if r.get('covered'):
            continue
        kind = r.get('kind')
        if kind in ('knockout', 'years'):
            block.append(f"mandatory: {r['requirement']}")
        elif kind == 'specialty':
            if r.get('years_have') is None:
                review.append(f"{r['requirement']}: {r.get('why', 'years unknown')}")
            else:
                block.append(f"{r['requirement']}: {r.get('why', 'not evidenced')}")
    # an empty tier-1 list is a review only when it looks like extraction
    # failed (years named in the JD and none extracted; a near-empty JD). A
    # JD whose only hard requirement is a Bachelor's he holds legitimately
    # has nothing left at tier 1.
    if not t1 and match.get('extraction_suspect'):
        review.append(f"no mandatory requirement extracted: {match['extraction_suspect']}")
    # title family he has not held (iam / implementation / clinical_ops):
    # not a rejection by itself — it passes when the JD names a specialty
    # his record proves, and goes to review when it names none
    fam = lane.get('family')
    domain = FAMILY_DOMAIN.get(fam, set())
    spec_ok = [r for r in reqs if r.get('kind') == 'specialty' and r.get('covered')
               and (not domain or set(r['requirement'].lower().replace('-', ' ').split()) & domain)]
    if fam not in HELD_FAMILIES and fam != 'other' and not spec_ok and not block:
        review.append(f"{fam} title is not a family he has held and the JD names no {fam} specialty his record proves")
    if score is not None:
        if score.get('fallback') or score.get('reason') == 'Could not score':
            review.append('match score is the fallback, not an assessment')
    if block:
        v = 'BLOCK'
    elif review:
        v = 'REVIEW'
    else:
        v = 'PASS'
    return {'verdict': v, 'block': block, 'review': review, 'reasons': block + review}

    return {'BLOCK': BLOCK_STATUS, 'REVIEW': REVIEW_STATUS}.get(v['verdict'])

### dry_run_boundary.py
```python
#!/usr/bin/env python3
"""The dry-run boundary: a dry run never touches a real employer account.

10/16: a Mac dry run of BMC 1137 passed the gate, opened the Workday page,
tried the stored password, had it rejected and clicked "Forgot password" —
a real reset email from BostonMedicalCenter at 1:44 AM. --dry-run stopped
the SUBMIT and nothing else. From here the boundary sits in the browser
layer, not in each handler's memory:

  live      every request allowed (dry_run False)
  dry       --dry-run: GET/HEAD/OPTIONS only, and no GET to a sign-in,
            account, password-reset, verification or OTP URL. Filling
            fields on a page is local; posting them, signing in, creating
            an account, recovering a password or requesting a code is a
            request, and the route guard aborts it before it leaves.
  dry-auth  --dry-run --auth (or SCIPIO_DRY_RUN_AUTH=1 with --dry-run):
            sign-in to an EXISTING account is allowed so an authenticated
            flow can be walked; account creation, password reset,
            verification and submission stay blocked. Never implied by
            --dry-run alone.

Two layers, same rules:
  install(context)      context.route('**/*') aborts blocked requests
                        (call at every place a browser context is made)
  guard(action)         raises DryRunBoundary inside the shared helpers
                        (sign-in, create account, reset, verification,
                        code polling, submit) so a handler cannot reach
                        the network step in the first place
"""
import os
import re
from urllib.parse import urlsplit

_MODE = 'live'
_SEEN: set = set()
SAFE_METHODS = {'GET', 'HEAD', 'OPTIONS'}
# URL shapes of account state changes. Sign-in has its own list so dry-auth
# can allow it while the rest stays shut.
SIGNIN_RX = re.compile(r'(sign[-_]?in|login|log[-_]?in|authenticate|session|oauth|sso|saml)', re.I)
ACCOUNT_RX = re.compile(r'(create[-_]?account|register|sign[-_]?up|reset[-_]?password|resetpassword|forgot|password[-_]?reset|'
                        r'recover|verif(y|ication)|activat(e|ion)|otp|one[-_]?time|security[-_]?code|mfa|2fa|confirm[-_]?email)', re.I)
SUBMIT_RX = re.compile(r'(/submit|/apply\b|/applications?\b|/candidates?\b|/postapply|/jobs?/[^/]+/apply)', re.I)
ACTIONS_LIVE_ONLY = {'create_account', 'reset_password', 'verification', 'security_code', 'submit'}
ACTIONS_AUTH_OK = {'signin'}


class DryRunBoundary(RuntimeError):
    """Raised when a dry run reaches a state-changing step."""


def set_mode(dry_run: bool, auth: bool | None = None) -> str:
    """Resolve and pin the mode for this process. auth is only honored
    together with dry_run; without --dry-run there is no dry-auth."""
    global _MODE
    if auth is None:
        auth = os.environ.get('SCIPIO_DRY_RUN_AUTH') == '1'
    _MODE = 'live' if not dry_run else ('dry-auth' if auth else 'dry')
    _SEEN.clear()
    return _MODE


def mode() -> str:
    return _MODE


def decide(method: str, url: str, current: str | None = None) -> tuple[bool, str]:
    """(allowed, reason) for one request under the mode."""
    m = (current or _MODE)
    if m == 'live':
        return True, 'live'
    method = (method or 'GET').upper()
    path = urlsplit(url or '').path + '?' + (urlsplit(url or '').query or '')
    if ACCOUNT_RX.search(path):
        return False, 'account state change'
    if SUBMIT_RX.search(path) and method not in SAFE_METHODS:
        return False, 'submission'
    if SIGNIN_RX.search(path) and m == 'dry':
        return False, 'sign-in'
    if method not in SAFE_METHODS:
        if m == 'dry-auth' and SIGNIN_RX.search(path):
            return True, 'dry-auth sign-in'
        if m == 'dry-auth' and not SIGNIN_RX.search(path) and not SUBMIT_RX.search(path) and not ACCOUNT_RX.search(path):
            # an authenticated walk needs the ATS's own page-render calls
            return True, 'dry-auth page traffic'
        return False, f'{method} in a dry run'
    return True, 'read'


def guard(action: str, what: str = '') -> None:
    """Raise unless the mode allows this action. Call BEFORE the step."""
    if _MODE == 'live':
        return
    if action in ACTIONS_AUTH_OK and _MODE == 'dry-auth':
        return
    raise DryRunBoundary(f'dry run ({_MODE}): refusing {action}' + (f' — {what}' if what else ''))


async def _route(route):
    req = route.request
    ok, why = decide(req.method, req.url)
    if ok:
        await route.continue_()
        return
    key = (req.method, urlsplit(req.url).netloc + urlsplit(req.url).path)
    if key not in _SEEN:
        _SEEN.add(key)
        print(f'  [dry] blocked {req.method} {req.url[:100]} ({why})')
    await route.abort('blockedbyclient')


async def install(context) -> bool:
    """Arm the request guard on a Playwright BrowserContext (launched,
    persistent or CDP-attached alike). No-op when live. Returns whether
    the guard is armed."""
    if _MODE == 'live':
        return False
    await context.route('**/*', _route)
    print(f'  [dry] boundary armed ({_MODE}): no sign-in, account, reset, verification or submit requests leave this browser')
    return True
```
### Remaining paths that can reach a live employer account
- Live mode (no --dry-run): everything, by design.
- `--dry-run --auth`: sign-in POST to an existing account only; create/reset/verification/submit still blocked at the request layer.
- Non-browser outbound: mailer.py (SMTP, has its own dry_run), captcha.py (2captcha API, called only when not dry_run and the browser guard still blocks the submit), scout/jd_extract/aggregators (GET job boards), state_lock (git), text_bridge (iMessage).
- dia_pdf.py / tailor.py open a browser only to render local HTML to PDF.
- The URL-pattern lists (SIGNIN_RX / ACCOUNT_RX / SUBMIT_RX) are heuristics for GETs; the hard rule is that no non-GET request leaves the browser in dry mode.
